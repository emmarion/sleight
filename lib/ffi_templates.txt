; Templates for the extraction of extern types (ASSUMPTIONS 133): "match T: text" for a type and
; "build T i: text" for its constructor i, text being OCaml over Obj.t with holes {s} the scrutinee,
; {bI} the body of constructor I's arm, {fI.J} the binder of its field J, {aI} a build argument and {tK}
; a temporary. Every hole but a body is filled with a variable name, and a body sits in tail position.
match Nat: if (Obj.obj {s} : Z.t) == Z.zero then {b0} else let {f1.0} = Obj.repr (Z.pred (Obj.obj {s} : Z.t)) in {b1}
build Nat 0: Obj.repr Z.zero
build Nat 1: Obj.repr (Z.succ (Obj.obj {a0} : Z.t))
match Half: let {t0} = (Obj.obj {s} : Z.t) in if {t0} == Z.zero then {b0} else let {t1} = (if Obj.is_int {s} then Obj.repr (((Obj.obj {s} : int) - 1) asr 1) else Obj.repr (Z.shift_right (Z.pred {t0}) 1)) in if (if Obj.is_int {s} then (Obj.obj {s} : int) land 1 = 1 else Z.is_odd {t0}) then let {f1.0} = {t1} in {b1} else let {f2.0} = {t1} in {b2}
build Half 0: Obj.repr Z.zero
build Half 1: Obj.repr (Z.succ (Z.shift_left (Obj.obj {a0} : Z.t) 1))
build Half 2: Obj.repr (Z.add (Z.shift_left (Obj.obj {a0} : Z.t) 1) (Z.of_int 2))
match String: let ({t0}, {t1}, {t2}) = (Obj.obj {s} : Ffi_core.slice) in if {t2} = 0 then {b0} else let {f1.0} = Obj.repr (Z.of_int (Char.code {t0}.[{t1}])) and {f1.1} = Obj.repr (({t0}, {t1} + 1, {t2} - 1) : Ffi_core.slice) in {b1}
build String 0: Obj.repr (("", 0, 0) : Ffi_core.slice)
build String 1: let ({t0}, {t1}, {t2}) = (Obj.obj {a1} : Ffi_core.slice) in Obj.repr ((String.make 1 (Char.chr (Z.to_int (Obj.obj {a0} : Z.t))) ^ String.sub {t0} {t1} {t2}, 0, {t2} + 1) : Ffi_core.slice)
